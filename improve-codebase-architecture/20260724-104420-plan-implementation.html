function createTestRunner(name) {
  const results = [];
  const ok = (name) => { results.push({ name, passed: true }); console.log('  ✅', name); };
  const fail = (name, err) => { results.push({ name, passed: false, error: err }); console.log('  ❌', name, '-', err); };
  return { results, ok, fail, report: () => {
    const passed = results.filter(r => r.passed).length;
    const failed = results.filter(r => !r.passed).length;
    console.log(`\n📊 ${name}: ${passed} ✅ / ${failed} ❌ / ${results.length} total`);
    return { total: results.length, passed, failed, results };
  }};
}

function escapeHtml(str) {
  if (!str) return '';
  return String(str)
    .replace(/&/g, '&')
    .replace(//g, '>')
    .replace(/"/g, '"');
}
